% Link-Farben: stabile ETH-Slots aus der Index-Palette (Blau/Petrol),
% unabhängig von der aktiven Kapitelfarbe.
\hypersetup{colorlinks=true, linkcolor=ZSFLinkColor, urlcolor=ZSFUrlColor}

% Build-Kennung unten mittig, Seitenzahl unten rechts (overlay).
% Lokal automatisch DEV + Datum; im Release-Workflow der SemVer-Tag.
%
% Das Overlay zeichnet ausserhalb des Textflusses und reserviert keine Höhe.
% Der Satzspiegel muss ihm den Platz deshalb selbst freihalten, sonst
% überdruckt der Footer die letzte Zeile einer voll gefüllten Spalte — was
% erst auffällt, sobald eine ZSF dicht genug gesetzt ist, um eine Spalte
% wirklich auszufüllen. Höhe wird aus der Footer-Schrift gemessen statt
% geraten, damit sie einer Schriftänderung folgt.
\settoheight{\ZSFfooterClearance}{\ZSFfontFooter 0}
\addtolength{\ZSFfooterClearance}{\ZSFfooterBottomInset}
\addtolength{\ZSFfooterClearance}{\ZSFspaceS}
\geometry{bottom=\ZSFfooterClearance}

\AddToShipoutPictureFG{%
  \AtPageLowerLeft{%
    \put(\dimexpr\paperwidth/2\relax,\ZSFfooterBottomInset){%
      \makebox(0,0)[b]{{\transparent{\ZSFfooterOpacity}\ZSFfontFooter\color{ReleaseFooterColor}\ZSFReleaseLabel}}%
    }%
    \put(\dimexpr\paperwidth-\ZSFfooterRightInset\relax,\ZSFfooterBottomInset){%
      \makebox(0,0)[rb]{\ZSFfontFooter\color{ZSFPageNumberColor}\thepage}%
    }%
  }%
}

% Enge Math-Display-Abstände (wie Analysis: keine grossen Lücken um Formeln)
\AtBeginDocument{
  \setlength{\abovedisplayskip}{\ZSFspaceXS}
  \setlength{\belowdisplayskip}{\ZSFspaceXS}
  \setlength{\abovedisplayshortskip}{\ZSFspaceXS}
  \setlength{\belowdisplayshortskip}{\ZSFspaceXS}
  \setlength{\mathsurround}{\ZSFspaceXS}
}

% Kompakte Listen (wie Analysis: kein überschüssiges Whitespace)
\setlist{nosep, itemsep=0pt, parsep=0pt, topsep=\ZSFspaceXS, partopsep=0pt}
\setlist[itemize]{leftmargin=\ZSFitemizeLeftMargin}
\setlist[enumerate]{leftmargin=\ZSFenumerateLeftMargin}

% Tabellen-Zellenabstand einheitlich
\setlength{\tabcolsep}{\ZSFspaceM}
\setlength{\extrarowheight}{\ZSFtextMinPad}
