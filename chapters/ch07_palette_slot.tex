\SubsectionBar[sec:code-listing]{CodeExpert-Listing}

\begin{codebox}[Was der Listing-Stil einfärbt\ZSFTitleTag{codebox}]
\begin{lstlisting}[style=CodeExpert]
# Kommentar: gruen gesetzt
def hervorhebung(text="Zeichenkette", zahl=42):
    if zahl > 0:
        return text
    return None
\end{lstlisting}
\end{codebox}

\SubsectionBar[sec:code-split]{Teilbare Code-Gruppe}

\begin{codeboxfirst}[Teil 1 trägt den Titel\ZSFTitleTag{codeboxfirst}]
\begin{lstlisting}[style=CodeExpert]
# Rahmen oben, unten offen
werte = [1, 2, 3]
\end{lstlisting}
\end{codeboxfirst}
\begin{codeboxmid}
\begin{lstlisting}[style=CodeExpert]
# codeboxmid: beide Kanten offen
summe = sum(werte)
\end{lstlisting}
\end{codeboxmid}
\begin{codeboxlast}
\begin{lstlisting}[style=CodeExpert]
# codeboxlast: Rahmen unten
print(summe)
\end{lstlisting}
\end{codeboxlast}
