\ZSFindex{figbox}

\begin{figbox}[Diagramm direkt im Dokument\ZSFTitleTag{figbox + pgfplots}]
  \centering
  \begin{tikzpicture}
    \begin{axis}[
      width=\linewidth,
      height=\ZSFfigMaxHeight,
      axis lines=middle,
      xmin=-2.2, xmax=2.2,
      ymin=-0.3, ymax=4.5,
      samples=40,
      ticks=none
    ]
      \addplot[color=\chaptercolor, very thick, domain=-2:2]{x^2};
      % Relative Achsenkoordinaten statt absoluter: die Beschriftung bleibt
      % dadurch garantiert innerhalb der Box und trifft die freie Fläche
      % zwischen den Parabelästen, statt am Rand abgeschnitten zu werden.
      \node[\chaptercolor, font=\ZSFfontDiagramLabel, anchor=center]
        at (rel axis cs:0.68,0.78) {ZSFfontDiagramLabel};
      % Zweite Stufe: die gedrängte Zusatzangabe, die neben der regulären
      % Marke stehen muss, ohne sie zu verdecken.
      \node[MutedTextColor, font=\ZSFfontDiagramLabelSmall, anchor=center]
        at (rel axis cs:0.68,0.62) {ZSFfontDiagramLabelSmall};
    \end{axis}
  \end{tikzpicture}
  \ZSFfigcaption{Zwei Beschriftungsstufen statt lokalem \textbackslash tiny:
    die reguläre Marke und die gedrängte Zusatzangabe. Für ein ganzes Bild
    lässt sich die Stufe einmal als font-Option der tikzpicture setzen.}
\end{figbox}

\begin{defbox}[Derselbe Regler, mit Kopfzeile\ZSFTitleTag{split}][split=0.34]
  \centering Formel
  \tcblower
  \ZSFkeyword*{split} ist ein Regler wie jeder andere und gilt deshalb auch
  auf einer Box mit Titel — vorher brauchte es dafür eine Verschachtelung.
\end{defbox}

\begin{splitbox}[split=0.34, splitalign=center]
  \centering linker Block
  \tcblower
  \ZSFkeyword{splitbox} ordnet zwei beliebige Inhaltsblöcke ohne sichtbaren
  Rahmen nebeneinander an; \ZSFkeyword*{split} legt den linken Anteil fest,
  \ZSFkeyword*{splitalign} die Ausrichtung der Hälften zueinander.
\end{splitbox}

\begin{splitbox}[split=0.34, splitalign=bottom]
  \centering linker Block
  \tcblower
  Dieselbe Box auf Grundlinie statt Mitte. Formel neben Erklärung will meist
  die Mitte, Diagramm neben Legende die Oberkante.
\end{splitbox}
